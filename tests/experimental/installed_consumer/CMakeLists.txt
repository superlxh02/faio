cmake_minimum_required(VERSION 4.1)
project(faio_installed_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()
if(DEFINED FAIO_CONSUMER_REJECT_COMPONENT)
    find_package(faio CONFIG REQUIRED COMPONENTS ${FAIO_CONSUMER_REJECT_COMPONENT})
    message(FATAL_ERROR "Unexpectedly accepted rejected component")
endif()

find_package(faio CONFIG REQUIRED)
add_executable(base base.cpp)
target_link_libraries(base PRIVATE faio::faio)
add_test(NAME base COMMAND base)
get_target_property(_faio_base_includes faio::faio INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(_faio_base_options faio::faio INTERFACE_COMPILE_OPTIONS)
if("${_faio_base_includes};${_faio_base_options}" MATCHES "stdexec|freflection")
    message(FATAL_ERROR "Base target unexpectedly carries experimental options")
endif()

find_package(faio CONFIG REQUIRED COMPONENTS experimental_runtime)
add_executable(runtime runtime.cpp)
target_link_libraries(runtime PRIVATE faio::experimental_runtime)
faio_configure_entry(runtime MODE NORMAL)
add_test(NAME runtime COMMAND runtime)

if(FAIO_CONSUMER_REFLECTION)
    find_package(faio CONFIG REQUIRED COMPONENTS experimental_reflection)
    add_executable(reflection reflection.cpp)
    target_link_libraries(reflection PRIVATE faio::experimental_reflection)
    faio_configure_entry(reflection MODE NORMAL)
    add_test(NAME reflection COMMAND reflection)
    add_executable(async_main async_main.cpp)
    faio_configure_entry(async_main MODE ASYNC)
    add_test(NAME async_main COMMAND async_main)
endif()

if(FAIO_CONSUMER_EXECUTION)
    find_package(faio CONFIG REQUIRED COMPONENTS experimental_execution)
    add_executable(execution execution.cpp)
    target_link_libraries(execution PRIVATE faio::experimental_execution)
    faio_configure_entry(execution MODE NORMAL)
    add_test(NAME execution COMMAND execution)
    get_target_property(_faio_execution_options faio::experimental_execution INTERFACE_COMPILE_OPTIONS)
    if(_faio_execution_options MATCHES "freflection")
        message(FATAL_ERROR "Execution target unexpectedly requires reflection")
    endif()
endif()
set_tests_properties(base PROPERTIES TIMEOUT 30)
