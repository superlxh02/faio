function(faio_add_module_example target component)
    add_executable("${target}" ${ARGN})
    set_property(TARGET "${target}" PROPERTY CXX_SCAN_FOR_MODULES ON)
    target_link_libraries("${target}" PRIVATE "${component}")
endfunction()

function(faio_register_module_test target)
    add_test(NAME "${target}" COMMAND "${target}" ${ARGN})
    set_tests_properties("${target}" PROPERTIES LABELS module TIMEOUT 60)
endfunction()

    function(faio_add_import_only_test component)
        set(_faio_target "faio_module_${component}_import_only")
        faio_add_module_example("${_faio_target}" faio::module import_only.cpp)
        string(TOUPPER "${component}" _faio_component_define)
        target_compile_definitions("${_faio_target}" PRIVATE
            "FAIO_IMPORT_MODULE=faio.${component}" "FAIO_IMPORT_${_faio_component_define}=1")
        if(NOT component MATCHES "^(common|log)$")
            target_compile_definitions("${_faio_target}" PRIVATE FAIO_IMPORT_HAS_TASK=1)
        endif()
        if(component MATCHES "^(runtime|tls|process)$")
            target_compile_definitions("${_faio_target}" PRIVATE FAIO_IMPORT_HAS_RUNTIME=1)
        endif()
        faio_register_module_test("${_faio_target}")
    endfunction()

    faio_add_module_example(faio_module_import_only faio::module import_only.cpp)

    faio_register_module_test(faio_module_import_only)
    foreach(_faio_component common coroutine io net fs runtime sync time log)
